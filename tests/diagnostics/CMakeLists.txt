cmake_minimum_required(VERSION 3.15)
project(yertle_diagnostics_test LANGUAGES C)
add_executable(firmware_diagnostics_test diagnostics_test.c ../../firmware/src/uart_diagnostics.c)
target_include_directories(firmware_diagnostics_test PRIVATE . ../../firmware/include)
if(MSVC)
    target_compile_options(firmware_diagnostics_test PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(firmware_diagnostics_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME firmware_diagnostics_test COMMAND firmware_diagnostics_test)
add_executable(firmware_dma_test dma_test.c ../../firmware/src/motor_dma.c
    ../../firmware/src/motor_uart.c ../../firmware/src/motor_device.c
    ../../firmware/src/uart_diagnostics.c ../../third_party/stm32-sts3215-lib/src/sts3215_protocol.c)
target_include_directories(firmware_dma_test PRIVATE . ../../firmware/include ../../third_party/stm32-sts3215-lib/inc)
target_compile_definitions(firmware_dma_test PRIVATE DMA_DRIVER_TEST=1)
if(MSVC)
    target_compile_options(firmware_dma_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
else()
    target_compile_options(firmware_dma_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME firmware_dma_test COMMAND firmware_dma_test)
