cmake_minimum_required(VERSION 3.24)
project(YertleCompareLauncher LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows launcher only; run compare_real_sim.py directly on other hosts")
endif()
get_filename_component(YERTLE_COMPARE_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/../../compare_real_sim.py" ABSOLUTE)
set(YERTLE_COMPARE_PYTHON "D:/tools/isg/e/Scripts/python.exe" CACHE FILEPATH "Existing Isaac Python")
configure_file(config.hpp.in generated/config.hpp @ONLY)
add_executable(yertle_compare main.cpp)
target_compile_features(yertle_compare PRIVATE cxx_std_17)
target_include_directories(yertle_compare PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(yertle_compare PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(yertle_compare PRIVATE shell32)
if(MSVC)
  target_compile_options(yertle_compare PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_test(NAME windows_argument_roundtrip COMMAND yertle_compare --self-test)
