cmake_minimum_required(VERSION 3.15)
project(yertle_pwm_control_tests LANGUAGES C)
enable_testing()
add_executable(pwm_control_test control_test.c ../../firmware/src/pwm_control.c ../../firmware/src/position_pid.c)
target_include_directories(pwm_control_test PRIVATE . ../../firmware/include)
if(MSVC)
 target_compile_options(pwm_control_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
else()
 target_compile_options(pwm_control_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME pwm_control_test COMMAND pwm_control_test)
add_executable(pwm_host_integration_test host_integration_test.c ../../firmware/src/actuator_task.c ../../firmware/src/pwm_control.c ../../firmware/src/position_pid.c)
target_compile_definitions(pwm_host_integration_test PRIVATE HOST_PROTOCOL_TEST=1)
target_include_directories(pwm_host_integration_test PRIVATE . ../safe_stop ../diagnostics ../../firmware/include)
if(MSVC)
 target_compile_options(pwm_host_integration_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
else()
 target_compile_options(pwm_host_integration_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME pwm_host_integration_test COMMAND pwm_host_integration_test)
add_executable(position_pid_test position_pid_test.c ../../firmware/src/position_pid.c)
target_include_directories(position_pid_test PRIVATE ../../firmware/include)
if(MSVC)
 target_compile_options(position_pid_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
else()
 target_compile_options(position_pid_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 target_link_libraries(pwm_control_test PRIVATE m)
 target_link_libraries(pwm_host_integration_test PRIVATE m)
 target_link_libraries(position_pid_test PRIVATE m)
endif()
add_test(NAME position_pid_numerical COMMAND position_pid_test)
add_executable(pwm_contract_test ../safe_stop/pwm_contract_test.c)
target_include_directories(pwm_contract_test PRIVATE ../../firmware/include)
add_test(NAME position_pid_wire_contract COMMAND pwm_contract_test)

foreach(target pwm_control_test pwm_host_integration_test)
 target_sources(${target} PRIVATE ../../firmware/src/mode_store.c mode_memory.c)
endforeach()
add_executable(mode_store_test mode_store_test.c mode_memory.c ../../firmware/src/mode_store.c)
target_include_directories(mode_store_test PRIVATE . ../../firmware/include)
if(MSVC)
 target_compile_options(mode_store_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
else()
 target_compile_options(mode_store_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME persistent_mode_journal COMMAND mode_store_test)

add_executable(calibration_store_test calibration_store_test.c calibration_memory.c ../../firmware/src/calibration_store.c)
target_include_directories(calibration_store_test PRIVATE ../../firmware/include)
if(MSVC)
 target_compile_options(calibration_store_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
else()
 target_compile_options(calibration_store_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME persistent_calibration_journal COMMAND calibration_store_test)
